## Опиание первой версии
Это система для сервиса по ремонту техники для организации учёта растрат на ремонты, в будущем она должна иметь возможность перерости в самостоятельную СРМ систему.

На данный момент сервис уже использует СРМ систему Ro App. Но она не позволяет удобно вести учет растрат на каждую заявку. Клиент хочет отдельный сервис который будет по апи связан с теперешней срм-кой, подтягивать оттуда заяви и считать растраты.

Основной функционал на теперь

Система должна иметь список заявок, склад, аналитику, систему пользователей и ролей.  так же должна быть как-то оформлена логика создания закупок. например что добавляется 10л масла на склад, потом из них списывается н-ное количество на какую-то заявку, или же закупили 5 карбюраторов, потом один например списали на одну заявку и помимо этого на заявку заказали еще свечу. Тоесть закупки нужно что-бы можно было делать как на склад так и на прямую в заявку. Помимо этого нужно сделать так что-бы если на склад приходит одна единица продукта но по разной цене, это учитывалось  а так же чтобы если продукт на складе заканчивался он не исчезал а был просто не в наличии и его можно было бы пополнить. отдельное замечание что б продукт на складе имел возможность прикрепления ссылок например с интернет магазинов и тд.

структуры данных

- Заявка
- Техника
- Клиент
- Пользователь
- Продукт
- Вложение

ниже приведен пример данных (это только на вскидку)

```
enum OrderStatus {
  inQueue = 'inQueue',
  inWork = 'inWork',
  Done = 'Done',
}

enum OrderType {
   Payed = 'Платний',
   Warranty = 'Гарантійний',
   Internal = "Внутрішній",
   UnPaid = "Безоплатний"
}

interface Photo {
   id: string,
   url: string,
   createdAt: string
}

interface Brand {
   id: string,
   title: string,
   photo: Photo
}

interface Client {
   id: string,                // ID
   isOrganization: boolean,   // Является ли организацией или компанией, или просто физ. лицо
   firstName: string,         // Имя
   lastName: string,          // фамилия
   email:string,              // Почта
   phone: string[]            // Номер телефона (может быть несколько)


}

interface Allocation {
   id: string;
   orderId: string;
   
   productId: string;        // ID товара в справочнике (например, "Карбюратор Honda Dio")
   productName: string;      // Название для удобного отображения в UI
   
   quantity: number;         // Количество (например, 1 шт. или 2 литра)
   unitPrice: number;        // Цена за единицу (автоматически подтягивается по FIFO из партии закупки)
   totalPrice: number;       // Общая стоимость (quantity * unitPrice)
   
   createdAt: Date;          // Когда мастер привязал запчасть к заказу
   createdById: string;      // Какой именно мастер выполнил списание
}

interface Asset {
   id: string
   uid: string,
   title: string
   type: string,
   brandId: string,
   model: string,
   modification: string,
}



interface Order {
   id: string                 // ID
   certificateId: string,     // номер ведомости

   createdById: string,       // ID кем создана заявка

   managerId: string,         // ID менеджера
   engineerId: string,        // ID исполнителя

   status: OrderStatus,       // статус заявки
   type: OrderType,           // тип заявки

   createdAt: Date,           // дата создания заявки
   modifiedAt: Date,          // дата последнего изменения заявки
   doneAt: Date,              // дата готовности
   closedAt: Date             // дата закрытия заявки

   dueDate: Date              // срок сдачи

   photoes: string[]          // идентефикаторы фоток 

   assetId: string,           // ID техники
   clientId: string,          // ID клиента

   malfunction: string,       // неисправность
   managerNotes: string,      // заметки менеджера
   engineerNotes: string,     // заметки исполнителя

   accsessories: string,      // комплектация

   allocation: Allocation[]   // вложения

   estimatedPrice: number,    // ориентировочная сума
   total: number              // конечная сума
}
```
